import type { JSX } from 'idlip/jsx-runtime';

type Save = { state: 'idle' } | { state: 'saving' } | { state: 'failed'; error: string };

export default function Saver(): JSX.Element {
  let save = { state: 'idle' } satisfies Save as Save;

  function start() {
    save = { state: 'saving' };
  }

  return (
    <div>
      #switch {
        #case (save.state === 'saving') {
          <p>Saving</p>
        }
        #case (save.state === 'failed') {
          <p class="error">{save.error}</p>
          <button on:click={start}>Retry</button>
        }
        #default {
          <button on:click={start}>Save</button>
        }
      }
    </div>
  );
}
