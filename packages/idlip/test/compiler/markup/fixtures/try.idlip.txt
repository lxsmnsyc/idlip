import type { JSX } from 'idlip/jsx-runtime';
import { Chart } from './chart.idlip';

export default function Report(props: { rows: number[] }): JSX.Element {
  return (
    <main>
      #try {
        <Chart rows={props.rows} />
      } #catch (error, reset) {
        <p>
          The chart failed: {String(error)}
          <button on:click={reset}>Retry</button>
        </p>
      }
    </main>
  );
}
