component TodoApp (stateful, exported)
  props: props
  state:
    draft = ""
    todos = props.initial
    next = (props.initial.length + 1)
  derived:
    left = countLeft(todos)
  handlers:
    setDraft(text)
      free: -
      writes: draft
    clear()
      free: -
      writes: draft
    add()
      free: draft:state todos:state next:state
      writes: todos next draft
    toggle(id)
      free: todos:state
      writes: todos

        " left"

        for #3 key=id item=t

            <input type="checkbox" checked={#5} [on:click -> toggle(t.id)  render reads todos.*.id]>
            {#6}
  holes:
    #0 draft
        expr  draft
        paths draft
    #1 draft === ''

        expr  countLeft(todos)

        paths todos

    #6 t.text
