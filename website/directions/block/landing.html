Features

  • MVU

    The model is plain data. Handlers write it and the view reads it.

    todos = [...todos, todo]

  • Resumability

    The server sends HTML and the model. No view code runs on load.

    <script i-state>

  • TSX function components

    let is state, const is derived, a function is a handler, return is the view.

    export default function TodoApp()

  • Subscriptions

    A timer or socket is a node in the view. It runs while its block is on the page.

    <Subscribe to={timer} />

  • Effects

    An await in a handler. The compiler splits the handler at the await.

    user = await loadUser(id)

  • Nesting

    A child keeps its own model. A click inside it loads only its chunks.

    <Counter start={0} />

  • Vite plugin

    One chunk per handler, binding group and list. Server pages in vite dev.

    plugins: [idlip()]

The whole todo app

tsx 
import { For } from 'idlip';
import type { JSX } from 'idlip/jsx-runtime';

interface Todo {
  id: string;
  text: string;
  done: boolean;
}

function countLeft(todos: readonly Todo[]): number {
  let left = 0;
  for (const todo of todos) {
    if (!todo.done) {
      left++;
    }
  }
  return left;
}

export default function TodoApp(props: { initial: Todo[] }): JSX.Element {
  let draft = '';
  let todos = props.initial;
  let next = props.initial.length + 1;

  const left = countLeft(todos);

  function setDraft(text: string) {
    draft = text;
  }
  function clear() {
    draft = '';
  }
  function add() {
    if (draft === '') {
      return;
    }
    todos = [...todos, { id: `t${next}`, text: draft, done: false }];
    next = next + 1;
    draft = '';
  }
  function toggle(id: string) {
    todos = todos.map((t) => (t.id === id ? { ...t, done: !t.done } : t));
  }

  return (
    <main>
      <form
        on:submit={(e) => {
          e.preventDefault();
          add();
        }}
      >
        <input
          value={draft}
          on:input={(e) => setDraft(e.currentTarget.value)}
          on:keydown={(e) => {
            if (e.key === 'Escape') {
              clear();
            }
          }}
        />
        <button disabled={draft === ''}>Add</button>
      </form>
      <p>{left} left</p>
      <ul>
        <For each={todos} key="id">
          {(t) => (
            <li class={{ done: t.done }}>
              <input type="checkbox" checked={t.done} on:click={() => toggle(t.id)} />
              {t.text}
            </li>
          )}
        </For>
      </ul>
    </main>
  );
}
